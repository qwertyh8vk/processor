potential_error assemble_default_validation(const char* command, long long command_number) {
    size_t length_of_command = strlen(command);
    PRINTF("<%s> command, default, debug\n", command);
    GETCHAR();

    if (length_of_command <= 4 && command[0] == ':') {
        char* end_ptr;
        long  label_index_in_array = strtol((command + 1 * sizeof(char)), &end_ptr, 10); // вопрос есть щас узнаем

        if (label_index_in_array > 0 && label_index_in_array < MAX_LABELS_AMOUNT - 1) {
            PRINTF("Индекс метки корреткный, DEFAULT, допустимый\n");
            PRINTF("Индекс метки в массиве: [%ld]\n", label_index_in_array);
            GETCHAR();

            return OK;
        }


        else {
            PRINTF("индекс метки некорректный, дропаю ассемблирование\n");
            GETCHAR();

            return INCORRECT_LABEL;
        }
    }

    else {
        PRINTF("=======================================================\n");
        PRINTF("<%lld> как я это запаковал, проверка в else\n" 
                "<%s> что это, проверка в else\n", command_number, command);
        PRINTF("=======================================================\n");

        PRINTF("Нераспознанное имя командлета или функции\n");
        GETCHAR();

        return UNKNOWN_COMMANDLETTE;
    }
} // целиком функция отладки, она ничего не делает кроме принтфов и ретурна если команда неизвестная
